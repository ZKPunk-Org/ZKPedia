<script>
(function () {
  try {
    var path = window.location.pathname;
    var isEn = path.indexOf('/en/') === 0;
    var h1 = document.querySelector('main h1');
    var body = document.getElementById('zkpedia-404-body');
    var links = document.getElementById('zkpedia-404-links');
    if (!body || !links) { return; }

    if (isEn) {
      // Requested an English page that has not been translated yet.
      if (h1) { h1.textContent = 'Not translated yet'; }
      document.title = 'Not translated yet - ZKPedia';
      document.documentElement.lang = 'en';   // picks up the English typography
      var zh = path.slice(3);
      body.textContent =
        'This page has no English version yet — or it never existed. ' +
        'ZKPedia is translated article by article, and this one is still on the list.';
      links.innerHTML =
        '<a href="' + zh + '">Read it in Chinese</a> &nbsp;·&nbsp; ' +
        '<a href="/en/">English home</a> &nbsp;·&nbsp; ' +
        '<a href="/en/how-to-contribute.html">Help translate it</a>';
    } else {
      var en = '/en' + path;
      links.innerHTML =
        '<a href="/">回到 ZKPedia 首页</a> &nbsp;·&nbsp; ' +
        '<a href="' + en + '">Try the English edition</a>';
    }
  } catch (e) {}
})();
</script>
